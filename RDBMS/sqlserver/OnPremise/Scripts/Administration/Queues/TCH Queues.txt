use CS;
set nocount on
--select GETUTCDATE() as UTCDateTime
select 
  GETUTCDATE() as UTCDateTime
  , TransactionsToVerify.TransactionStatusID, 
  TransactionVerificationStatus.Description
  , Qty = Count(TransactionsToVerify.TransactionID) 
 from TransactionsToVerify (nolock)
  inner join TransactionVerificationStatus (nolock) on
   TransactionsToVerify.TransactionStatusID = TransactionVerificationStatus.VerificationStatusID
 --where TransactionsToVerify.TransactionStatusID in (120, 130, 160 )
 group by 
  TransactionsToVerify.TransactionStatusID, TransactionVerificationStatus.Description
 order by TransactionStatusID
